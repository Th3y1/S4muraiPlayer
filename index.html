async function addAnime() { const pass = document.getElementById('admin-pass').value; const title = document.getElementById('anime-title').value; const category = document.getElementById('anime-category').value; const image = document.getElementById('anime-image').value; const episodes = document.getElementById('anime-episodes').value; if (pass !== 'Korneliusz1111') { alert('Błędne hasło administratora!'); return; } if (!title || !episodes) { alert('Wypełnij przynajmniej tytuł oraz link do odcinka!'); return; } try { const response = await fetch(`${SUPABASE_URL}/rest/v1/S4muraiPlayer`, { method: 'POST', headers: { 'apikey': SUPABASE_KEY, 'Authorization': `Bearer ${SUPABASE_KEY}`, 'Content-Type': 'application/json', 'Prefer': 'return=minimal' }, body: JSON.stringify({ title, category, image, episodes }) }); if (!response.ok) throw new Error('Błąd zapisu: ' + response.status); alert('Anime zostało dodane pomyślnie!'); document.getElementById('anime-title').value = ''; document.getElementById('anime-image').value = ''; document.getElementById('anime-episodes').value = ''; document.getElementById('admin-pass').value = ''; fetchAnime(); } catch (err) { alert('Błąd podczas dodawania: ' + err.message); } }
